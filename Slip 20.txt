using System;

class Program
{
    static void Main()
    {
        Console.WriteLine("Perfect numbers between 1 and 100 are:");

        for (int num = 1; num <= 100; num++)
        {
            int sum = 0;

            // Find divisors of num
            for (int i = 1; i <= num / 2; i++)
            {
                if (num % i == 0)
                {
                    sum += i;
                }
            }

            // Check perfect number condition
            if (sum == num && num != 0)
            {
                Console.WriteLine(num);
            }
        }

        Console.ReadLine();
    }
}

using System;

class Program
{
    static void Main()
    {
        Console.Write("Enter number of rows (M): ");
        int m = int.Parse(Console.ReadLine());

        Console.Write("Enter number of columns (N): ");
        int n = int.Parse(Console.ReadLine());

        int[,] matrix = new int[m, n];

        Console.WriteLine("Enter matrix elements:");

        // Input matrix
        for (int i = 0; i < m; i++)
        {
            for (int j = 0; j < n; j++)
            {
                Console.Write($"Element [{i},{j}]: ");
                matrix[i, j] = int.Parse(Console.ReadLine());
            }
        }

        int lowerSum = 0;
        int upperSum = 0;

        // Calculate lower and upper triangle sums
        for (int i = 0; i < m; i++)
        {
            for (int j = 0; j < n; j++)
            {
                if (i > j)
                {
                    lowerSum += matrix[i, j];   // Lower triangle
                }
                else if (i < j)
                {
                    upperSum += matrix[i, j];   // Upper triangle
                }
            }
        }

        Console.WriteLine("\nSum of Lower Triangular Elements = " + lowerSum);
        Console.WriteLine("Sum of Upper Triangular Elements = " + upperSum);

        Console.ReadLine();
    }
}
