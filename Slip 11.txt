using System;
using System.Collections.Generic;
using System.Linq;

class Program
{
    static void Main()
    {
        Stack<int> stack = new Stack<int>();

        // Push elements
        stack.Push(10);
        stack.Push(20);
        stack.Push(30);
        stack.Push(40);

        Console.WriteLine("Stack Elements: " + string.Join(", ", stack));

        // Top element
        Console.WriteLine("Top Element: " + stack.Peek());

        // Bottom element
        Console.WriteLine("Bottom Element: " + stack.Last());
    }
}

using System;

// Interface
interface Gross
{
    double TA { get; }
    double DA { get; }

    double Gross_sal();
}

// Base Class
class Employee
{
    protected string name;
    protected double basic_sal;

    public Employee(string n, double b)
    {
        name = n;
        basic_sal = b;
    }
}

// Derived Class implementing Interface
class Salary : Employee, Gross
{
    private double HRA;

    public Salary(string n, double b, double h) : base(n, b)
    {
        HRA = h;
    }

    public double TA
    {
        get { return 800; }
    }

    public double DA
    {
        get { return 500; }
    }

    public double Gross_sal()
    {
        return basic_sal + HRA + TA + DA;
    }

    public void Disp_sal()
    {
        Console.WriteLine("Employee Name : " + name);
        Console.WriteLine("Basic Salary  : " + basic_sal);
        Console.WriteLine("HRA           : " + HRA);
        Console.WriteLine("TA            : " + TA);
        Console.WriteLine("DA            : " + DA);
        Console.WriteLine("Gross Salary  : " + Gross_sal());
    }
}

// Main Class
class Program
{
    static void Main()
    {
        Salary s = new Salary("Omkar", 25000, 4000);
        s.Disp_sal();
    }
}
