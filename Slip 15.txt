using System;
using System.Drawing;
using System.Windows.Forms;

public class Form1 : Form
{
    Label lbl;
    Timer t;

    public Form1()
    {
        this.Text = "Scrolling Text";
        this.Size = new Size(500, 200);

        lbl = new Label();
        lbl.Text = "Welcome to C# Programming";
        lbl.AutoSize = true;
        lbl.Location = new Point(0, 70);
        this.Controls.Add(lbl);

        t = new Timer();
        t.Interval = 100;   // speed
        t.Tick += new EventHandler(MoveText);
        t.Start();
    }

    void MoveText(object sender, EventArgs e)
    {
        lbl.Left += 5;   // move text left to right

        if (lbl.Left > this.Width)
            lbl.Left = -lbl.Width;   // restart from left
    }

    [STAThread]
    static void Main()
    {
        Application.Run(new Form1());
    }
}

using System;
using System.Data.SqlClient;

class Program
{
    static void Main()
    {
        SqlConnection con = new SqlConnection(
            @"Data Source=.;Initial Catalog=TestDB;Integrated Security=True");

        con.Open();

        // a) Create Table
        SqlCommand cmd = new SqlCommand(
            "create table Student(RollNo int, Name varchar(30))", con);
        cmd.ExecuteNonQuery();
        Console.WriteLine("Table Created");

        // b) Alter Table
        cmd = new SqlCommand(
            "alter table Student add City varchar(20)", con);
        cmd.ExecuteNonQuery();
        Console.WriteLine("Table Altered");

        // c) Drop Table
        cmd = new SqlCommand("drop table Student", con);
        cmd.ExecuteNonQuery();
        Console.WriteLine("Table Dropped");

        con.Close();
    }
}
